# ============================================================================

#
# Единственный файл конфигурации. Правьте его и перезапускайте приложение —
# больше нигде ничего менять не нужно.
#
# Формат: КЛЮЧ=ЗНАЧЕНИЕ, по одному на строку.
#   # в начале строки — комментарий;
#   # после пробела — тоже комментарий: OLLAMA_MODEL=qwen2.5:7b-instruct # так можно
#   ключи регистронезависимы, значения сохраняют регистр.
#
# Если ключ не указан — берётся значение по умолчанию (оно указано в скобках).

# ----------------------------------------------------------------------------
# Адрес приложения
# ----------------------------------------------------------------------------
# На каком интерфейсе сервер принимает подключения.
#   127.0.0.1 — только с этой машины (по умолчанию, безопаснее)
#   0.0.0.0   — доступен по сети: http://<ip-компьютера>:<APP_PORT>
APP_HOST=127.0.0.1

# Порт HTTP-сервера. Фронтенд и API отдаются на одном порту, отдельно
# настраивать бэкенд не нужно. (8080)
APP_PORT=8080

# Куда браузер обращается к API.
#   пусто      — тот же адрес, откуда открыт интерфейс (обычный случай)
#   например  — http://192.168.1.50:8080, если фронт живёт на другой машине
# Пустое значение — оставить как есть.
API_BASE_URL=


# ----------------------------------------------------------------------------
# Модель
# ----------------------------------------------------------------------------
# Адрес локальной модели Ollama (OpenAI-совместимый API).
# Модель должна быть скачана заранее:  ollama pull <OLLAMA_MODEL>
# По умолчанию http://localhost:11434
OLLAMA_BASE_URL=http://localhost:11434

# Модель, которую использует приложение. Должна существовать в Ollama:
#   ollama list      — посмотреть установленные
#   ollama pull X    — скачать
# По умолчанию qwen2.5:7b-instruct
OLLAMA_MODEL=qwen2.5:7b-instruct

# Таймаут одного запроса к модели, секунд. Увеличьте для больших моделей
# на медленных процессорах. (30)
AI_TIMEOUT_SEC=30


# ----------------------------------------------------------------------------
# Журнал
# ----------------------------------------------------------------------------
# Уровень логов: fatal, error, warn, info, debug, trace, silent. (info)
LOG_LEVEL=info

# Путь к файлу лога. Пусто — писать только в окно консоли.
LOG_FILE=


# ----------------------------------------------------------------------------
# Прочее
# ----------------------------------------------------------------------------
# Открывать браузер сразу после запуска. true / false. (true)
OPEN_BROWSER=true
